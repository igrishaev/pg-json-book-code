prepare flatten as
with recursive step as (
    select
        jt.*,
        0 as level
    from
        json_table($1::jsonb, '$' columns(
            id int4 path '$.id',
            code text path '$.code',
            children jsonb path '$.children'
    )) as jt
    UNION ALL
    select
        jt.*,
        step.level + 1 as level
    from
        step,
        json_table(step.children, '$[*]' columns(
            id int4 path '$.id',
            code text path '$.code',
            children jsonb path '$.children'
    )) as jt)
select id, code, level from step
order by level;
